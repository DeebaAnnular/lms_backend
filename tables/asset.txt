CREATE TABLE assets (
    asset_id INT AUTO_INCREMENT PRIMARY KEY,
    asset_no VARCHAR(255) UNIQUE,
    asset_type VARCHAR(255),
    brand_name VARCHAR(255),
    device_serial_number VARCHAR(255),
    asset_status TINYINT,
    ram VARCHAR(255),
    rom VARCHAR(255),
    processor VARCHAR(255),
    ostype VARCHAR(255),
    ms_office_installed TINYINT,
    os_installed TINYINT,
    purchase_date DATE,
    purchase_cost DECIMAL(10, 2),
    admin_configuration TINYINT,
    operational_status VARCHAR(255),
    command VARCHAR(255),
    user_id INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
);

CREATE TABLE asset_maintenance (
    asset_service_id INT AUTO_INCREMENT PRIMARY KEY,
    asset_no VARCHAR(50) NOT NULL,
    issue_description TEXT NOT NULL,
    service_cost DECIMAL(10, 2) NOT NULL,
    service_outdate DATE NOT NULL,
    service_indate DATE NOT NULL
);
